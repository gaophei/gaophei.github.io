#现场备注(2026-07-15):
#1)OS路径大小写敏感:第1步实建目录为小写(stuwork/portal/dataassets),DIRECTORY路径必须与OS完全一致;
#首版误按大写建立,已用create or replace改指小写并复核如上;
#2)create or replace directory保留既有对象授权——Oracle SQL Reference明确OR REPLACE的用途即
#"重定义目录而无需drop/重建/重新授权";重复grant幂等无害(v2初版此处误记为"会清空授权",已勘误)。
#复核: select grantee,privilege from dba_tab_privs where table_name='EXPDP_DIR';
#若改由SYSTEM自建目录则创建者自动获得读写,此时不要再grant给自己(报ORA-01749);
#3)核查语句原名单中PORTALUSER不存在(0行),CON_ID 4实际账号为PORTALADMIN/PORTAL_SERVICE_V6,
#名单已按实盘修正并重跑留痕;
#4)整改项(不阻塞备份,列台账):PORTALADMIN账号EXPIRED(GRACE)且默认表空间为SYSTEM;DSADMIN默认表空间为SYSTEM。
#同属"应用管理账号落SYSTEM表空间"的不规范配置;本备份统一以SYSTEM执行,不受影响;
#5)master table建于执行账号SYSTEM的默认表空间(即SYSTEM表空间),体量极小、job结束自动清理,剩余空间已确认;

  local node="$1" free pct
  free=$(rsh "${node}" "timeout 15 df -BG --output=avail '${BASE_DIR}'" 2>/dev/null | awk 'NR==2{gsub("G","");print $1}')
  pct=$(rsh "${node}" "timeout 15 df -P '${BASE_DIR}'" 2>/dev/null | awk 'NR==2{gsub("%","");print $5}')
  if [ -z "${free}" ] || [ -z "${pct}" ]; then SPACE_MSG="${node}: df failed or timed out"; return 1; fi
  if [ "${free}" -lt "${MIN_FREE_GB}" ] || [ "${pct}" -gt "${MAX_USED_PCT}" ]; then
    SPACE_MSG="${node}: free=${free}G(min ${MIN_FREE_GB}G) used=${pct}%(max ${MAX_USED_PCT}%)"; return 1
  fi
  SPACE_MSG="${node}: free=${free}G used=${pct}%"; return 0
}
